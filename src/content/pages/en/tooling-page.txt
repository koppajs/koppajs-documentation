            <h3>configuration</h3>
            <pre class="docs-code"><code>import { defineConfig } from "vite";
import koppaPlugin from "@koppajs/koppajs-vite-plugin";

export default defineConfig({
  plugins: [
    koppaPlugin({
      tsconfigFile: "./tsconfig.json",
    }),
  ],
});</code></pre>

      <section id="typescript" class="docs-section-card docs-section-anchor">
        <p class="docs-eyebrow">TypeScript</p>
        <h2 class="docs-section-title">Recommended compiler settings and <code>.kpa</code> typing.</h2>
        <div class="docs-grid-2">
          <article class="docs-mini-card">
            <h3><code>tsconfig.json</code></h3>
            <pre class="docs-code"><code>{
  "compilerOptions": {
    "target": "ES2022",
    "module": "ESNext",
    "lib": ["ES2022", "DOM"],
    "strict": true,
    "moduleResolution": "bundler",
    "esModuleInterop": true,
    "skipLibCheck": true,
    "forceConsistentCasingInFileNames": true,
    "noEmit": true
  },
  "include": ["src/**/*.ts"],
  "exclude": ["node_modules", "dist"]
}</code></pre>
          </article>

          <article class="docs-mini-card">
            <h3><code>.kpa</code> imports</h3>
            <pre class="docs-code"><code>import myComponent from "./my-component.kpa";</code></pre>
            <p class="docs-text">The core library automatically provides the necessary type declarations for <code>.kpa</code> modules. You don’t need any additional manual module definition for this.</p>
          </article>
        </div>

      <section id="debugging" class="docs-section-card docs-section-anchor">
        <p class="docs-eyebrow">Loggers &amp; Debugging</p>
        <h2 class="docs-section-title">Built-in logging for development and production.</h2>
        <div class="docs-grid-2">
          <article class="docs-mini-card">
            <h3>initialization</h3>
            <pre class="docs-code"><code>import { logger, LogLevel } from "@koppajs/koppajs-core";

logger.init({
  level: LogLevel.DEBUG,
  isDevelopment: true,
});

logger.setLevel(LogLevel.DEBUG);</code></pre>
          </article>

          <article class="docs-mini-card">
            <h3>Log level</h3>
            <div class="docs-table-wrap">
              <table class="docs-table">
                <thead>
                  <tr>
                    <th>level</th>
                    <th>Value</th>
                    <th>Description</th>
                  </tr>
                </thead>
                <tbody>
                  <tr><td><code>DEBUG</code></td><td>0</td><td>All news</td></tr>
                  <tr><td><code>INFO</code></td><td>1</td><td>Info, warning and error</td></tr>
                  <tr><td><code>WARN</code></td><td>2</td><td>Warn and error</td></tr>
                  <tr><td><code>ERROR</code></td><td>3</td><td>Just mistakes</td></tr>
                  <tr><td><code>NONE</code></td><td>4</td><td>No output</td></tr>
                </tbody>
              </table>
            </div>
            <ul class="docs-list">
              <li>Development starts by default on <code>DEBUG</code>.</li>
              <li>Production defaults to <code>ERROR</code>.</li>
            </ul>
          </article>
        </div>
      </section>

      <section id="editor" class="docs-section-card docs-section-anchor">
        <p class="docs-eyebrow">VS Code</p>
        <h2 class="docs-section-title">Editor support for clean work in single-file components.</h2>
        <article class="docs-mini-card">
          <ul class="docs-list">
            <li>Syntax highlighting for <code>.kpa</code> files</li>
            <li>Detection of the blocks <code>[template]</code>, <code>[ts]</code>/<code>[js]</code> and <code>[css]</code>/<code>[scss]</code>/<code>[sass]</code></li>
            <li>HTML highlighting in the template block</li>
            <li>TypeScript or JavaScript highlighting in the script block</li>
            <li>CSS or SCSS highlighting in the style block</li>
          </ul>
        </article>
